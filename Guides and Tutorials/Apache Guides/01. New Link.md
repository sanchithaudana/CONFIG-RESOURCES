## New

### New Config
```bash
DOMAIN=iwmi-pv.minchy.ai
FOLDER=iwmi_pv
CONF=/etc/apache2/sites-available/$DOMAIN.conf
```

```bash
[ -e "$CONF" ] && echo "$CONF already exists" || sudo tee "$CONF" > /dev/null <<EOF
<VirtualHost *:80>
        ServerName $DOMAIN
        DocumentRoot "/var/www/html/$FOLDER"

        ErrorLog  "/var/log/apache2/$FOLDER-error.log"
        CustomLog "/var/log/apache2/$FOLDER-access.log" combined

        <Directory "/var/www/html/$FOLDER">
                Options FollowSymLinks
                AllowOverride All
                Require all granted
        </Directory>

        <Directory "/var/www/html/$FOLDER/.git">
                Require all denied
        </Directory>
</VirtualHost>
EOF
```

```bash
sudo a2ensite "$DOMAIN"
sudo apache2ctl configtest
sudo systemctl reload apache2
```

### HTTPS

#### Let's encrypt

```bash
sudo apt install certbot python3-certbot-apache -y
```

```bash
sudo certbot --apache -d "$DOMAIN"
```

#### Custom
```bash
SSLCONF=/etc/apache2/sites-available/$DOMAIN-ssl.conf

[ -e "$SSLCONF" ] && echo "$SSLCONF already exists" || sudo tee "$SSLCONF" > /dev/null <<EOF
<IfModule mod_ssl.c>
<VirtualHost *:443>
        ServerName $DOMAIN
        DocumentRoot "/var/www/html/$FOLDER"

        ErrorLog  "/var/log/apache2/$FOLDER-https-error.log"
        CustomLog "/var/log/apache2/$FOLDER-https-access.log" combined

        SSLEngine on
        SSLCertificateFile /etc/letsencrypt/live/$DOMAIN/fullchain.pem
        SSLCertificateKeyFile /etc/letsencrypt/live/$DOMAIN/privkey.pem
        Include /etc/letsencrypt/options-ssl-apache.conf

        <Directory "/var/www/html/$FOLDER">
                Options FollowSymLinks
                AllowOverride All
                Require all granted
        </Directory>

        <Directory "/var/www/html/$FOLDER/.git">
                Require all denied
        </Directory>
</VirtualHost>
</IfModule>
EOF
```

```bash
sudo a2ensite "$DOMAIN-ssl"
sudo apache2ctl configtest 
sudo systemctl reload apache2
```

## Append
### Append Vhost
```bash
sudo tee -a /etc/apache2/sites-available/vhosts.conf > /dev/null <<'EOF'

<VirtualHost *:80>
        ServerName <DOMAIN>
        DocumentRoot "/var/www/html/<FOLDER_NAME>"

        ErrorLog  "/var/log/apache2/<FOLDER_NAME>-error.log"
        CustomLog "/var/log/apache2/<FOLDER_NAME>-access.log" combined

        <Directory "/var/www/html/<FOLDER_NAME>">
                Options FollowSymLinks
                AllowOverride All
                Require all granted
        </Directory>

        <Directory "/var/www/html/<FOLDER_NAME>/.git">
                Require all denied
        </Directory>
</VirtualHost>
EOF
```
